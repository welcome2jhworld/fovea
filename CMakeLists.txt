cmake_minimum_required(VERSION 3.25)
project(fovea VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake")

option(FOVEA_BUILD_CONSOLE "Build the Qt Widgets desktop console" ON)
option(FOVEA_BUILD_TOOLS "Build local test tools (RTSP test source)" ON)
option(FOVEA_BUILD_TESTS "Build unit tests" ON)

find_package(Qt6 6.5 REQUIRED COMPONENTS Core Network Sql HttpServer)
if(FOVEA_BUILD_CONSOLE)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Gui Widgets Svg)
endif()
if(FOVEA_BUILD_TESTS)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
  enable_testing()
endif()
qt_standard_project_setup()
include(FoveaGStreamer)
include(FoveaWarnings)

add_subdirectory(src/common)
add_subdirectory(src/rules)
add_subdirectory(src/core)
if(FOVEA_BUILD_CONSOLE)
  add_subdirectory(src/console)
endif()
if(FOVEA_BUILD_TOOLS AND GST_RTSP_SERVER_FOUND)
  add_subdirectory(tools/rtsp-testsrc)
endif()
if(FOVEA_BUILD_TESTS)
  add_subdirectory(tests)
endif()
