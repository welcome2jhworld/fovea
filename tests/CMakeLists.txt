function(fovea_add_test name)
  qt_add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE fovea-common Qt6::Test)
  fovea_target_warnings(${name})
  add_test(NAME ${name} COMMAND ${name})
endfunction()
